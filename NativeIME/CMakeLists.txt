cmake_minimum_required(VERSION 3.25)
project(NativeIME VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(nativeime_core
    src/core/pinyin_engine.cpp
    src/core/app_policy.cpp
    src/core/engine_protocol.cpp
)
target_include_directories(nativeime_core PUBLIC src)
target_compile_features(nativeime_core PUBLIC cxx_std_17)

if(MSVC)
    target_compile_options(nativeime_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(nativeime_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(nativeime_engine_protocol_tests tests/engine_protocol_tests.cpp)
target_link_libraries(nativeime_engine_protocol_tests PRIVATE nativeime_core)
enable_testing()
add_test(NAME nativeime_engine_protocol_tests COMMAND nativeime_engine_protocol_tests)

add_executable(nativeime_core_tests tests/core_tests.cpp)
target_link_libraries(nativeime_core_tests PRIVATE nativeime_core)

add_executable(nativeime_session_tests tests/session_tests.cpp)
target_sources(nativeime_core PRIVATE src/core/input_session.cpp)
target_link_libraries(nativeime_session_tests PRIVATE nativeime_core)
if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(nativeime_core PRIVATE /utf-8)
    target_compile_options(nativeime_core_tests PRIVATE /utf-8)
    target_compile_options(nativeime_engine_protocol_tests PRIVATE /utf-8)
    target_compile_options(nativeime_session_tests PRIVATE /utf-8)
endif()

add_test(NAME nativeime_core_tests COMMAND nativeime_core_tests)

add_test(NAME nativeime_session_tests COMMAND nativeime_session_tests)

add_executable(nativeime_app_policy_tests tests/app_policy_tests.cpp)
target_link_libraries(nativeime_app_policy_tests PRIVATE nativeime_core)
add_test(NAME nativeime_app_policy_tests COMMAND nativeime_app_policy_tests)

option(NATIVEIME_WITH_RIME "Build isolated real librime candidate provider" OFF)
set(NATIVEIME_RIME_DATA_DIR "" CACHE PATH "Path to installed Rime schema data")
set(NATIVEIME_RIME_RUNTIME_DLL "" CACHE FILEPATH "Absolute path to Windows rime.dll")
if(NATIVEIME_WITH_RIME)
    if(WIN32)
        set(RIME_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/librime/include")
    else()
        find_path(RIME_INCLUDE_DIR rime_api.h REQUIRED)
        find_library(RIME_LIBRARY NAMES rime librime REQUIRED)
    endif()
    add_library(nativeime_rime src/core/rime_provider.cpp)
    target_include_directories(nativeime_rime PRIVATE ${RIME_INCLUDE_DIR})
    target_link_libraries(nativeime_rime PUBLIC nativeime_core)
    if(WIN32)
        target_compile_definitions(nativeime_rime PRIVATE NATIVEIME_RIME_DYNAMIC)
    else()
        target_link_libraries(nativeime_rime PRIVATE ${RIME_LIBRARY})
    endif()
    add_executable(nativeime_rime_smoke tests/rime_smoke.cpp)
    target_link_libraries(nativeime_rime_smoke PRIVATE nativeime_rime)
    if(NATIVEIME_RIME_DATA_DIR)
        if(WIN32)
            if(NATIVEIME_RIME_RUNTIME_DLL)
                add_test(NAME nativeime_rime_smoke COMMAND nativeime_rime_smoke "${NATIVEIME_RIME_DATA_DIR}" "${CMAKE_BINARY_DIR}/rime-test-data" "${NATIVEIME_RIME_RUNTIME_DLL}")
            endif()
        else()
            add_test(NAME nativeime_rime_smoke COMMAND nativeime_rime_smoke "${NATIVEIME_RIME_DATA_DIR}" "${CMAKE_BINARY_DIR}/rime-test-data")
        endif()
    endif()
endif()

option(NATIVEIME_DISPOSABLE_REGISTRATION_TEST "Run real register/unregister only on disposable Windows CI" OFF)
option(NATIVEIME_EXPERIMENTAL_TSF "Enable unvalidated inline TSF composition for lab only" OFF)
if(WIN32)
    add_library(nativeime_tsf SHARED
        src/windows/tsf_dll.cpp
        src/windows/tsf_edit.cpp
        src/windows/NativeIME.def
    )
    target_compile_definitions(nativeime_tsf PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
    target_link_libraries(nativeime_tsf PRIVATE ole32 advapi32 uuid user32)
    if(NATIVEIME_EXPERIMENTAL_TSF)
        target_compile_definitions(nativeime_tsf PRIVATE NATIVEIME_EXPERIMENTAL_TSF)
        target_sources(nativeime_tsf PRIVATE src/windows/tsf_popup.cpp src/windows/focus_context.cpp)
        target_link_libraries(nativeime_tsf PRIVATE nativeime_core gdi32 nativeime_engine_client)
    endif()
    if(NATIVEIME_WITH_RIME)
        add_executable(NativeIME.EngineService src/windows/engine_service.cpp)
        target_link_libraries(NativeIME.EngineService PRIVATE nativeime_rime advapi32)
        target_compile_definitions(NativeIME.EngineService PRIVATE UNICODE _UNICODE)
        add_executable(NativeIME.EngineSupervisor src/windows/engine_supervisor.cpp)
        target_compile_definitions(NativeIME.EngineSupervisor PRIVATE UNICODE _UNICODE)
        target_link_libraries(NativeIME.EngineSupervisor PRIVATE advapi32)
    endif()
    add_library(nativeime_engine_client STATIC src/windows/engine_client.cpp)
    target_include_directories(nativeime_engine_client PUBLIC src)
    target_link_libraries(nativeime_engine_client PUBLIC nativeime_core advapi32)
    if(NATIVEIME_WITH_RIME AND NATIVEIME_RIME_DATA_DIR AND NATIVEIME_RIME_RUNTIME_DLL)
        add_executable(nativeime_ipc_integration tests/ipc_integration.cpp src/windows/tsf_popup.cpp)
        target_link_libraries(nativeime_ipc_integration PRIVATE nativeime_engine_client user32 gdi32 uuid)
        add_test(NAME nativeime_ipc_integration COMMAND nativeime_ipc_integration $<TARGET_FILE:NativeIME.EngineService> "${NATIVEIME_RIME_DATA_DIR}" "${CMAKE_BINARY_DIR}/ipc-rime-data" "${NATIVEIME_RIME_RUNTIME_DLL}")
        set_tests_properties(nativeime_ipc_integration PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
    endif()
    add_executable(nativeime_focus_tests tests/focus_context_tests.cpp src/windows/focus_context.cpp)
    target_link_libraries(nativeime_focus_tests PRIVATE nativeime_core user32)
    add_test(NAME nativeime_focus_tests COMMAND nativeime_focus_tests)
    if(NATIVEIME_DISPOSABLE_REGISTRATION_TEST)
        add_executable(nativeime_tsf_registration tests/tsf_registration.cpp)
        target_link_libraries(nativeime_tsf_registration PRIVATE ole32 oleaut32 uuid advapi32)
        add_test(NAME nativeime_tsf_registration COMMAND nativeime_tsf_registration $<TARGET_FILE:nativeime_tsf> --disposable-ci)
        set_tests_properties(nativeime_tsf_registration PROPERTIES RUN_SERIAL TRUE TIMEOUT 60)
    endif()
    add_executable(nativeime_tsf_smoke tests/tsf_smoke.cpp)
    target_link_libraries(nativeime_tsf_smoke PRIVATE ole32 uuid)
    add_test(NAME nativeime_tsf_smoke COMMAND nativeime_tsf_smoke $<TARGET_FILE:nativeime_tsf>)
    add_executable(NativeIME.DesktopLab WIN32 src/windows/desktop_lab.cpp)
    target_compile_definitions(NativeIME.DesktopLab PRIVATE UNICODE _UNICODE)
    target_link_libraries(NativeIME.DesktopLab PRIVATE user32 gdi32)
    add_executable(NativeIME.InputLab WIN32 src/windows/preview.cpp)
    target_compile_definitions(NativeIME.InputLab PRIVATE UNICODE _UNICODE)
    target_link_libraries(NativeIME.InputLab PRIVATE nativeime_core user32 gdi32)
    set_target_properties(nativeime_tsf PROPERTIES
        OUTPUT_NAME "NativeIME.TSF"
        CXX_VISIBILITY_PRESET hidden
    )
endif()
